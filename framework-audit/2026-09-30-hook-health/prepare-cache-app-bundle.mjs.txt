const sha=bytes=>createHash('sha256').update(bytes).digest('hex');
const hex=value=>typeof value==='string'&&/^[a-f0-9]{64}$/.test(value);
const fail=message=>{throw new Error(message);};
const normalize=value=>value&&typeof value==='object'?Array.isArray(value)?value.map(normalize):Object.fromEntries(Object.keys(value).sort().map(key=>[key,normalize(value[key])])):value;

function within(child,parent){const rel=path.relative(parent,child);return rel===''||(!rel.startsWith('..')&&!path.isAbsolute(rel));}
function safeName(name){return typeof name==='string'&&name&&!path.isAbsolute(name)&&!name.includes('\\')&&!name.includes('\0')&&!name.split('/').some(x=>!x||x==='.'||x==='..');}
function keys(value,expected){return value&&typeof value==='object'&&!Array.isArray(value)&&equal(Object.keys(value).sort(),[...expected].sort());}
function canonical(file){if(!path.isAbsolute(file)||path.resolve(file)!==file||fs.realpathSync(file)!==file)fail(`noncanonical path: ${file}`);return file;}
function readBytes(file){
  canonical(file);const before=fs.lstatSync(file);
  if(!before.isFile()||before.isSymbolicLink()||before.size>512*1024*1024)fail(`unsafe file: ${file}`);
  const fd=fs.openSync(file,fs.constants.O_RDONLY|(fs.constants.O_NOFOLLOW||0));
  try{
    const opened=fs.fstatSync(fd);
    if(!opened.isFile()||opened.dev!==before.dev||opened.ino!==before.ino)fail(`opening drift: ${file}`);
    const bytes=fs.readFileSync(fd),after=fs.fstatSync(fd),named=fs.lstatSync(file);
    if(bytes.length!==opened.size||after.size!==opened.size||after.mtimeMs!==opened.mtimeMs||after.dev!==opened.dev||after.ino!==opened.ino||named.dev!==opened.dev||named.ino!==opened.ino)fail(`reading drift: ${file}`);
    return bytes;
  }finally{fs.closeSync(fd);}
}
function run(command,args,options={}){
  const result=spawnSync(command,args,{encoding:'utf8',maxBuffer:32*1024*1024,stdio:['ignore','pipe','pipe'],...options});
  if(result.error||result.status!==0)fail(`${path.basename(command)} failed: ${(result.stderr||result.error?.message||'').trim()}`);
  return `${result.stdout||''}${result.stderr||''}`;
}

function verifyScope(scopeFile,scopeSha){
  canonical(scopeFile);
  if(!within(scopeFile,AUDIT)||scopeFile===AUDIT)fail('scope must be an absolute file below the exact audit directory');

  const scope=JSON.parse(bytes);
  if(!keys(scope,['schema_version','kind','sets','plan','hooks','install_review','pending_live'])||scope.schema_version!==1||scope.kind!=='hook-cache-final-file-set'||!Array.isArray(scope.sets)||scope.sets.length!==2)fail('scope schema mismatch');
  let hostFile=null;const currentHeads=[];
  for(const [index,set]of scope.sets.entries()){
    if(!keys(set,['root','base','files','diff_sha256'])||set.root!==[...BASES.keys()][index]||set.base!==BASES.get(set.root)||!hex(set.diff_sha256)||!Array.isArray(set.files)||set.files.length!==FILES[index].length)fail('scope root/base/file set mismatch');
    canonical(set.root);
    if(git(set.root,['rev-parse',`${set.base}^{commit}`]).trim()!==set.base)fail('scope base is not the exact baseline commit');
    git(set.root,['merge-base','--is-ancestor',set.base,'HEAD']);
    currentHeads.push({root:set.root,base:set.base,head:git(set.root,['rev-parse','HEAD']).trim()});
    for(const [i,row]of set.files.entries()){
      if(!keys(row,['path','relative','sha256','bytes','tracked'])||row.relative!==FILES[index][i]||row.path!==path.join(set.root,row.relative)||!hex(row.sha256)||!Number.isSafeInteger(row.bytes)||typeof row.tracked!=='boolean')fail('scope file record mismatch');
      const actual=readBytes(row.path);
      if(actual.length!==row.bytes||sha(actual)!==row.sha256)fail(`reviewed source drift: ${row.path}`);
      if(row.relative==='app/host-hook-ready.js')hostFile=row;
    }
    if(sha(git(set.root,['diff',set.base,'--',...set.files.filter(row=>row.tracked).map(row=>row.relative)]))!==set.diff_sha256)fail('reviewed diff SHA drift');
  }
  for(const name of ['plan','hooks','install_review']){
    const item=scope[name];
    if(!keys(item,['path','sha256'])||!hex(item.sha256)||!within(item.path,AUDIT)||sha(readBytes(item.path))!==item.sha256)fail(`scope ${name} drift`);
  }
  if(!hostFile)fail('scope lacks the host module');
  return {scope,bytes,hostFile,currentHeads};
}
function treeInventory(root){
  canonical(root);const records={};
  function visit(folder,rel){
    const stat=fs.lstatSync(folder);if(!stat.isDirectory()||stat.isSymbolicLink())fail('unsafe inventory directory');
    records[rel||'.']={type:'directory',mode:stat.mode&0o7777};
    for(const name of ordered(fs.readdirSync(folder))){
      const file=path.join(folder,name),key=rel?`${rel}/${name}`:name,entry=fs.lstatSync(file);
      if(!safeName(key))fail('unsafe app inventory name');
      if(entry.isDirectory())visit(file,key);
      else if(entry.isFile())records[key]={type:'file',mode:entry.mode&0o7777,size:entry.size,sha256:sha(readBytes(file))};
      else if(entry.isSymbolicLink()){
        const target=fs.readlinkSync(file);
        if(!within(path.resolve(path.dirname(file),target),root))fail(`app link escapes bundle: ${key}`);
        records[key]={type:'link',mode:entry.mode&0o7777,target};
      }else fail(`special app file: ${key}`);
    }
  }
  visit(root,'');return records;
}
function plist(file){return JSON.parse(run('/usr/bin/plutil',['-convert','json','-o','-',file]));}
function archiveInventory(asar,file){
  asar.uncache(file);const names=asar.listPackage(file),records=[];
  if(new Set(names).size!==names.length)fail('duplicate asar names');
  for(const name of names){
    if(!name.startsWith('/')||!safeName(name.slice(1)))fail('unsafe asar name');
    const relative=name.slice(1),meta=asar.statFile(file,relative,false);
    if('files'in meta)records.push({path:relative,type:'directory',unpacked:meta.unpacked===true,meta});
    else if('link'in meta){if(!safeName(meta.link))fail('unsafe asar link');records.push({path:relative,type:'link',unpacked:meta.unpacked===true,meta});}
    else{
      const content=asar.extractFile(file,relative,false);
      if(content.length!==meta.size)fail(`asar short read: ${relative}`);
      records.push({path:relative,type:'file',unpacked:meta.unpacked===true,sha256:sha(content),size:content.length,meta});
    }
  }
  return records;
}

}
function unchangedInstalled(asar,before){

  if(now.inventory_sha256!==before.inventory_sha256||now.asar_sha256!==before.asar_sha256||now.header_sha256!==before.header_sha256||now.info_sha256!==before.info_sha256||now.unpacked_sha256!==before.unpacked_sha256||!equal(now.sidecar,before.sidecar))fail('installed app/asar/plist/native/sidecar CAS drift');
}
function emptyBlockOnly(before,after){
  if(!before||!after)return false;
  const old={...before,blocks:undefined},next={...after,blocks:undefined};
  return equal(old,next)&&equal(before.blocks,[])&&equal(after.blocks,[sha(Buffer.alloc(0))]);
}
function compareArchives(before,after,hostSha){
  if(before.length!==after.length)fail('asar module count changed');
  const changes=[],metadata=[];
  for(let i=0;i<before.length;i++){
    const a=before[i],b=after[i];
    if(a.path!==b.path||a.type!==b.type||a.unpacked!==b.unpacked)fail(`asar order/type/unpacked flag changed: ${a.path}`);
    if(a.type==='file'){
      if(a.path===HOST){if(b.sha256!==hostSha)fail('prepared host module differs from reviewed source');if(a.sha256!==b.sha256)changes.push(a.path);}
      else if(a.sha256!==b.sha256||a.size!==b.size)fail(`unreviewed module changed: ${a.path}`);
      const old={...a.meta},next={...b.meta};delete old.offset;delete next.offset;
      if(a.path===HOST){delete old.size;delete next.size;delete old.integrity;delete next.integrity;}
      else if(a.size===0&&emptyBlockOnly(old.integrity,next.integrity)){metadata.push({path:a.path,reason:'official serializer normalizes empty blocks',before:old.integrity,after:next.integrity});delete old.integrity;delete next.integrity;}
      if(!equal(old,next))fail(`asar metadata/flags changed: ${a.path}`);
      if(b.meta.integrity?.algorithm!=='SHA256'||b.meta.integrity.hash!==b.sha256)fail(`asar integrity hash mismatch: ${a.path}`);
    }else{
      const old={...a.meta},next={...b.meta};delete old.files;delete next.files;
      if(!equal(old,next))fail(`asar directory/link flags changed: ${a.path}`);
    }
  }
  if(changes.length!==1||changes[0]!==HOST)fail('package must change exactly the reviewed host module');
  return {changed_modules:changes,allowed_metadata_changes:metadata,offsets_recomputed:true,modules_read_back:after.length};
}
function assertTemp(file,root){if(!within(file,root)||file===root)fail('write escaped sole temporary package root');}
function writeTemp(file,bytes,root){assertTemp(file,root);fs.writeFileSync(file,bytes,{flag:'wx',mode:0o600});}
function thinMachOPayload(bytes){
  // Resigning can change an embedded signature but may not change executable payload.
  if(bytes.length<32||bytes.readUInt32LE(0)!==0xfeedfacf)return null;
  const count=bytes.readUInt32LE(16),commands=bytes.readUInt32LE(20);if(count>10000||32+commands>bytes.length)return null;
  const normalized=Buffer.from(bytes);let offset=32,signature=null;
  for(let i=0;i<count;i++){
    if(offset+8>32+commands)return null;const command=bytes.readUInt32LE(offset),size=bytes.readUInt32LE(offset+4);
    if(size<8||offset+size>32+commands)return null;
    if(command===0x1d){if(size!==16||signature)return null;signature={start:bytes.readUInt32LE(offset+8),length:bytes.readUInt32LE(offset+12)};normalized.fill(0,offset+8,offset+16);}
    offset+=size;
  }
  if(!signature||signature.start+signature.length!==bytes.length||signature.start<32+commands)return null;
  return normalized.subarray(0,signature.start);
}
function verifyBundleChanges(before,after,staged){
  if(!equal(Object.keys(before),Object.keys(after)))fail('bundle file inventory changed');
  const signatureChanges=[];
  for(const name of Object.keys(before)){
    const a=before[name],b=after[name];if(equal(a,b))continue;
    if(a.type!==b.type||a.mode!==b.mode)fail(`bundle type/mode changed: ${name}`);
    if(name===ASAR_REL||name===INFO_REL)continue;
    if(name.includes('/_CodeSignature/')&&a.type==='file'){signatureChanges.push({path:name,before_sha256:a.sha256,after_sha256:b.sha256,kind:'signature seal'});continue;}
    if(a.type==='file'&&b.type==='file'&&!name.startsWith(`${UNPACKED_REL}/`)){
      const old=thinMachOPayload(readBytes(path.join(INSTALLED,name))),next=thinMachOPayload(readBytes(path.join(staged,name)));
      if(old&&next&&old.equals(next)){signatureChanges.push({path:name,before_sha256:a.sha256,after_sha256:b.sha256,kind:'embedded signature only'});continue;}
    }
    fail(`unreviewed bundle bytes changed: ${name}`);
  }
  return signatureChanges;
}
